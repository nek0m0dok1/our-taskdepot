cmake_minimum_required(VERSION 4.2)

if(NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026")
  message(FATAL_ERROR "This project supports only the Visual Studio 18 2026 generator (VS2026).")
endif()

project(our_taskdepot_foundation LANGUAGES C)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This foundation targets Windows x64 only.")
endif()
if(NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "The application foundation requires the VS2026 MSVC compiler.")
endif()
if(MSVC_VERSION LESS 1950)
  message(FATAL_ERROR "MSVC 19.50 or later from Visual Studio 2026 is required.")
endif()
if(NOT CMAKE_VS_PLATFORM_TOOLSET STREQUAL "v145")
  message(FATAL_ERROR "The VS2026 v145 platform toolset is required.")
endif()
if(NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION)
  message(FATAL_ERROR "A Windows SDK is required; install Windows 11 SDK 10.0.22000.0 or later.")
elseif(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.22000.0")
  message(FATAL_ERROR "Windows 11 SDK 10.0.22000.0 or later is required; CMake selected ${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}.")
endif()

include(CTest)
include(cmake/Dependencies.cmake)

if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
